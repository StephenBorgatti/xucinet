// --- Structural holes, ego-network model ------------------------------------
// holes(<net> [method]); the method is the egonet definition, union by
// default, which is also the dialog's default ("Union - either kind"). The
// command writes Term1-Term3 after the dialog's eleven columns; the tests use
// the first eleven. It loops over every relation; the tests read the first.
g8_holes_campnet  = holes(campnet)
g8_holes_hightech = holes(hightech)
g8_holes_sampson  = holes(sampson)
g8_holes_iso      = holes(g9_iso)
g8_holes_disc     = holes(g9_disc)

// --- Egonet alter composition, categorical ----------------------------------
// egocomp(<net> <attr> [type:BOTH|out|in|recip] [ignoreown:NO|yes]); the
// defaults are the dialog's.
g8_egocomp_campnet_gender = egocomp(campnet campattr$gender)

// --- Ego-alter similarity, categorical ---------------------------------------
// categohom(<net> <attr> [type:OUT|in|any|recip]) takes the first column of
// the attribute dataset, so pull Gender out first. Outgoing is the dialog's
// default too.
gen = col(campattr 1)
g8_egohom_campnet_gender = categohom(campnet gen type:outgoing)
