// ---------------------------------------------------------------------------
// PART A - the command line
// ---------------------------------------------------------------------------

// --- Reciprocity (Network | Cohesion | Reciprocity) ------------------------
// Dialog: Method = Hybrid is the default (pMethod.ItemIndex = 2), which at the
// whole-network level equals Dyad-based. Save all three so that claim can be
// checked rather than trusted, and save the node-level dataset too.
g10_recip_campnet_dyad = recip(campnet dyad)
g10_recip_campnet_arc  = recip(campnet arc)
g10_recip_sampson      = recip(sampson dyad)
// camp92 is valued: this is the one that shows the two halves of the report
// using different tests (ledger entry 16).
g10_recip_camp92       = recip(camp92 dyad)

// --- Transitivity (Network | Cohesion | Transitivity) ----------------------
// Dialog: Method = Triplets is the default (Method.ItemIndex = 1); run Triads
// as well, since the R function offers both.
g10_trans_campnet_trip = trans(campnet triplets)
g10_trans_campnet_tri  = trans(campnet triads)
g10_trans_hightech     = trans(hightech triplets)
g10_trans_disc         = trans(g9_disc triplets)

// --- Cyclicality (Network | Cohesion | Cyclicality) ------------------------
g10_cyc_campnet = cyclic(campnet)
g10_cyc_sampson = cyclic(sampson)

// --- Components (Network | Regions | Components) ---------------------------
// Dialog: "For directed data" = Weak Components by default
// (Method.ItemIndex = 0); sort frequencies by Component Size
// (sortby.ItemIndex = 1). Save the partition vector each time.
// g9_disc and g9_iso are the ones that matter: several components, and an
// isolate that must come back as a component of size 1.
g10_comp_campnet_weak   = comp(campnet weak)
g10_comp_campnet_strong = comp(campnet strong)
g10_comp_disc_weak      = comp(g9_disc weak)
g10_comp_iso_weak       = comp(g9_iso weak)
// Record from the log, for each run: Component ratio, Heterogeneity,
// Normalized heterogeneity, Entropy, Normalized entropy. Those five are
// printed and not saved to a dataset, so the log is the only record of them.
